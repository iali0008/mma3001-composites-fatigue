protocol_v2

Protocol v2 feature extraction and the predeclared selection rule.

Implements sections 2-4 and 7 of docs/protocol_v2.md with the settings in config/protocol_v2.json:

  • joint, label-blind recording eligibility (both input sets use the same recordings, checkpoints and reference);
  • the earliest eligible checkpoint as reference, chosen without labels;
  • the single_path_1_7 (v1 baseline) and multi_path_36 input sets;
  • the worst-direction-MAE selection rule with its tie-break order.

Labels are read only when read_labels=True. With read_labels=False the label field is deleted from each MAT structure as soon as it is loaded, so the held-out specimen can be processed without exposing its labels.

 77def summarise_file(coupon, specimen, filename, protocol, read_labels):
 78    """Collect metadata, settings and per-pair features of one MAT file.
 79
 80    Parameters
 81    ----------
 82    coupon : dict
 83        ``coupon`` structure of one MAT file.
 84    specimen : str
 85        Specimen name.
 86    filename : str
 87        MAT file name.
 88    protocol : dict
 89        Output of :func:`load_protocol`.
 90    read_labels : bool
 91        Read the published label; if false the label field is deleted first.
 92
 93    Returns
 94    -------
 95    dict
 96        File summary. ``pair_counts`` counts records per pair at the protocol
 97        frequency; ``settings`` maps pair to (gain, amplitude, sampling_rate);
 98        ``rms``/``ptp`` map pair to receiver features when every check for that
 99        pair passes; ``bad_pairs`` lists pairs with invalid signals.
100    """
101    if not read_labels:
102        coupon.pop('straingage_data', None)
103    rules = protocol['eligibility']
104    records = coupon.get('path_data', [])
105    records = [records] if isinstance(records, dict) else list(records)
106    counts, settings, rms, ptp, bad = Counter(), {}, {}, {}, []
107    for p in records:
108        if scalar(p.get('frequency', [])) != rules['frequency_khz']:
109            continue
110        pair = (scalar(p.get('actuator', [])), scalar(p.get('sensor', [])))
111        counts[pair] += 1
112        settings[pair] = (
113            scalar(p.get('gain', [])),
114            scalar(p.get('amplitude', [])),
115            scalar(p.get('sampling_rate', [])),
116        )
117        a = np.asarray(p.get('signal_actuator', []), dtype=float).ravel()
118        x = np.asarray(p.get('signal_sensor', []), dtype=float).ravel()
119        ok = all(
120            len(s) == rules['signal_length_samples'] and np.isfinite(s).all() and np.ptp(s) > 0
121            for s in (a, x)
122        )
123        if ok:
124            rms[pair], ptp[pair] = waveform_features(x)
125        else:
126            bad.append(pair)
127    label = None
128    if read_labels:
129        label = scalar(coupon.get('straingage_data', {}).get('stiffness_degradation', []))
130    return dict(
131        specimen=specimen,
132        filename=filename,
133        cycles=scalar(coupon.get('cycles', [])),
134        load_kips=scalar(coupon.get('load', [])),
135        condition=str(coupon.get('condition', '')).strip(),
136        published_index=label,
137        pair_counts=counts,
138        settings=settings,
139        rms=rms,
140        ptp=ptp,
141        bad_pairs=bad,
142    )

163def eligibility_reasons(summary, specimen_rules, modal_vector, protocol):
164    """List every protocol v2 rule a file fails; an empty list means eligible.
165
166    Parameters
167    ----------
168    summary : dict
169        Output of :func:`summarise_file`.
170    specimen_rules : dict
171        ``protocol['specimens'][specimen]``.
172    modal_vector : tuple
173        The specimen's modal :func:`setting_vector`.
174    protocol : dict
175        Output of :func:`load_protocol`.
176
177    Returns
178    -------
179    list of str
180        Failed rules. Labels are never consulted.
181    """
182    rules, reasons = protocol['eligibility'], []
183    if summary['cycles'] is None or summary['cycles'] <= 0:
184        reasons.append('not a positive-cycle checkpoint')
185    if summary['condition'].casefold() != 'loaded':
186        reasons.append('condition is not loaded')
187    low, high = specimen_rules['load_window_kips']
188    if summary['load_kips'] is None or not low <= summary['load_kips'] <= high:
189        reasons.append(f'load outside {low}-{high} kips')
190    grid = pairs(protocol)
191    if any(summary['pair_counts'].get(p, 0) != 1 for p in grid):
192        reasons.append('36-pair grid incomplete or duplicated')
193    base = (rules['baseline_path']['actuator'], rules['baseline_path']['sensor'])
194    if summary['pair_counts'].get(base, 0) != 1:
195        reasons.append('baseline path missing or ambiguous')
196    elif summary['settings'][base][:2] != (
197        specimen_rules['baseline_gain'],
198        specimen_rules['baseline_amplitude'],
199    ):
200        reasons.append('baseline gain/amplitude mismatch')
201    if any(
202        summary['settings'].get(p, (None,) * 3)[2] != rules['stored_sampling_rate'] for p in grid
203    ):
204        reasons.append('stored sampling rate mismatch')
205    if setting_vector(summary, protocol) != modal_vector:
206        reasons.append('setting vector differs from specimen modal vector')
207    if any(p in summary['bad_pairs'] or p not in summary['rms'] for p in grid):
208        reasons.append('invalid signal (length, nonfinite or constant)')
209    return reasons

212def extract_specimen(data_root, specimen, protocol, read_labels, workers=1):
213    """Screen every PZT file of one specimen with the protocol v2 rules.
214
215    Parameters
216    ----------
217    data_root : pathlib.Path
218        The extracted ``2. Composites`` directory.
219    specimen : str
220        Specimen name in ``protocol['specimens']``.
221    protocol : dict
222        Output of :func:`load_protocol`.
223    read_labels : bool
224        Whether labels may be read (``False`` for the held-out test specimen
225        until step 4).
226    workers : int, optional
227        MAT prefetch workers.
228
229    Returns
230    -------
231    summaries : list of dict
232        One summary per file with ``eligible`` and ``exclusion_reasons`` added.
233    hashes : dict
234        SHA-256 per file, keyed relative to ``data_root``.
235    """
236    rules = protocol['specimens'][specimen]
237    files = sorted((data_root / rules['folder'] / 'PZT-data').glob(specimen + '_*.mat'))
238    require(bool(files), f'No PZT files for {specimen}')
239    summaries, hashes = [], {}
240    for path, digest_value, coupon in iter_coupons(files, workers):
241        hashes[path.relative_to(data_root).as_posix()] = digest_value
242        summaries.append(summarise_file(coupon, specimen, path.name, protocol, read_labels))
243    modal = Counter(setting_vector(s, protocol) for s in summaries).most_common(1)[0][0]
244    for s in summaries:
245        reasons = eligibility_reasons(s, rules, modal, protocol)
246        s['eligible'], s['exclusion_reasons'] = not reasons, '; '.join(reasons)
247    return summaries, hashes

250def build_checkpoints(summaries, protocol):
251    """Aggregate eligible recordings into checkpoints with both input sets.
252
253    Parameters
254    ----------
255    summaries : list of dict
256        Output of :func:`extract_specimen` for one specimen.
257    protocol : dict
258        Output of :func:`load_protocol`.
259
260    Returns
261    -------
262    checkpoints : list of dict
263        Every checkpoint with at least one eligible recording, in cycle order,
264        with ``relative_rms``/``relative_ptp`` (pair 1 -> 7) and
265        ``mp_relative_rms``/``mp_relative_ptp`` (median of 36 per-pair ratios).
266        ``published_index`` is ``None`` when unlabelled or labels were not read.
267    pair_ratios : list of dict
268        Per-checkpoint, per-pair ratios behind the multi-path features.
269    reference : dict
270        Reference cycle, files and denominators.
271
272    Raises
273    ------
274    ValueError
275        If no recording is eligible, labels conflict within a checkpoint, or a
276        reference denominator is not positive.
277    """
278    eligible = [s for s in summaries if s['eligible']]
279    require(bool(eligible), 'No eligible recording')
280    specimen = eligible[0]['specimen']
281    groups = defaultdict(list)
282    for s in eligible:
283        groups[s['cycles']].append(s)
284    grid = pairs(protocol)
285    base = (
286        protocol['eligibility']['baseline_path']['actuator'],
287        protocol['eligibility']['baseline_path']['sensor'],
288    )
289    medians = {}
290    for cycle, group in groups.items():
291        medians[cycle] = {
292            p: (
293                float(np.median([s['rms'][p] for s in group])),
294                float(np.median([s['ptp'][p] for s in group])),
295            )
296            for p in grid
297        }
298    ref_cycle = min(groups)  # earliest eligible checkpoint; labels never consulted
299    ref = medians[ref_cycle]
300    require(all(v > 0 for p in grid for v in ref[p]), 'Invalid reference denominator')
301    reference = dict(
302        specimen=specimen,
303        cycles=int(ref_cycle),
304        filenames='; '.join(s['filename'] for s in groups[ref_cycle]),
305        rms_denominator=ref[base][0],
306        ptp_denominator=ref[base][1],
307    )
308    checkpoints, pair_ratios = [], []
309    for cycle in sorted(groups):
310        group = groups[cycle]
311        labels = {s['published_index'] for s in group if s['published_index'] is not None}
312        require(len(labels) <= 1, f'Conflicting published labels: {specimen}, {cycle}')
313        rms_ratio = {p: medians[cycle][p][0] / ref[p][0] for p in grid}
314        ptp_ratio = {p: medians[cycle][p][1] / ref[p][1] for p in grid}
315        checkpoints.append(
316            dict(
317                specimen=specimen,
318                cycles=int(cycle),
319                published_index=labels.pop() if labels else None,
320                eligible_repeats=len(group),
321                filenames='; '.join(s['filename'] for s in group),
322                load_min_kips=min(s['load_kips'] for s in group),
323                load_max_kips=max(s['load_kips'] for s in group),
324                is_reference=cycle == ref_cycle,
325                reference_cycles=int(ref_cycle),
326                mean_centred_rms=medians[cycle][base][0],
327                peak_to_peak=medians[cycle][base][1],
328                relative_rms=medians[cycle][base][0] / ref[base][0],
329                relative_ptp=medians[cycle][base][1] / ref[base][1],
330                mp_relative_rms=float(np.median(list(rms_ratio.values()))),
331                mp_relative_ptp=float(np.median(list(ptp_ratio.values()))),
332                log10_cycles=float(np.log10(cycle)),
333            )
334        )
335        for p in grid:
336            pair_ratios.append(
337                dict(
338                    specimen=specimen,
339                    cycles=int(cycle),
340                    actuator=int(p[0]),
341                    sensor=int(p[1]),
342                    rms_ratio=rms_ratio[p],
343                    ptp_ratio=ptp_ratio[p],
344                )
345            )
346    return checkpoints, pair_ratios, reference

367def select_final(candidates, protocol):
368    """Apply the predeclared worst-direction-MAE selection rule.
369
370    Parameters
371    ----------
372    candidates : list of dict
373        One row per (model, input_set) with ``model``, ``input_set`` and
374        ``worst_direction_MAE`` (MLP: five-seed median per direction, then max).
375    protocol : dict
376        Output of :func:`load_protocol`.
377
378    Returns
379    -------
380    dict
381        ``selected`` (the chosen row), ``minimum`` worst-direction MAE and the
382        ``tied`` candidates within the tolerance, in tie-break order.
383
384    Raises
385    ------
386    ValueError
387        If the candidate set is not exactly the 8 declared candidates.
388
389    Examples
390    --------
391    >>> p = {'selection_rule': {'tie_tolerance_mae': 0.0005, 'tie_break_order': {
392    ...     'model': list(FAMILIES), 'input_set': list(INPUT_SETS)}}}
393    >>> rows = [dict(model=m, input_set=i, worst_direction_MAE=0.1) for m in FAMILIES
394    ...         for i in INPUT_SETS]
395    >>> rows[-1]['worst_direction_MAE'] = 0.0996   # MLP multi-path, within tolerance
396    >>> select_final(rows, p)['selected']['model']
397    'LinearRegression'
398    """
399    rule = protocol['selection_rule']
400    require(
401        sorted((c['model'], c['input_set']) for c in candidates)
402        == sorted((m, i) for m in FAMILIES for i in INPUT_SETS),
403        'Unexpected candidate set',
404    )
405    best = min(c['worst_direction_MAE'] for c in candidates)
406    order_m, order_i = rule['tie_break_order']['model'], rule['tie_break_order']['input_set']
407    tied = sorted(
408        [c for c in candidates if c['worst_direction_MAE'] - best <= rule['tie_tolerance_mae']],
409        key=lambda c: (order_m.index(c['model']), order_i.index(c['input_set'])),
410    )
411    return dict(selected=tied[0], minimum=best, tied=tied)

414def read_published_labels(data_root, specimen, protocol, workers=1):
415    """Read only the published label of every PZT file of one specimen.
416
417    Used for the held-out specimen *after* its predictions have been written,
418    so features and predictions are fixed before any label is seen.
419
420    Parameters
421    ----------
422    data_root : pathlib.Path
423        The extracted ``2. Composites`` directory.
424    specimen : str
425        Specimen name in ``protocol['specimens']``.
426    protocol : dict
427        Output of :func:`load_protocol`.
428    workers : int, optional

431    Returns

447def attach_labels(checkpoints, labels):
448    """Copy published labels onto checkpoints from their eligible files.
449
450    Parameters
451    ----------
452    checkpoints : list of dict
453        Output of :func:`build_checkpoints` built with ``read_labels=False``.
454    labels : dict
455        Output of :func:`read_published_labels`.
456
457    Returns
458    -------
459    list of dict
460        Copies with ``published_index`` set (``None`` if no file is labelled).
461
462    Raises
463    ------
464    ValueError
465        If files of one checkpoint carry different labels or a file is unknown.
466    """
467    out = []
468    for cp in checkpoints:
469        names = cp['filenames'].split('; ')
470        require(all(n in labels for n in names), f"Unknown file at cycle {cp['cycles']}")
471        found = {labels[n][1] for n in names if labels[n][1] is not None}
472        require(len(found) <= 1, f"Conflicting published labels at cycle {cp['cycles']}")
473        out.append(dict(cp, published_index=found.pop() if found else None))
474    return out

477def sanity_criteria(late_predictions, prediction_at_100k, training_targets, protocol):
478    """Evaluate the predeclared plausibility criteria C1-C3 (protocol section 9).
479
480    Parameters
481    ----------
482    late_predictions : sequence of float
483        Frozen-model predictions at the unlabelled late L1S11 checkpoints.
484    prediction_at_100k : float
485        Frozen-model prediction at the L1S11 100,000-cycle checkpoint.
486    training_targets : sequence of float
487        Published targets of the pooled training checkpoints.
488    protocol : dict
489        Output of :func:`load_protocol` (only documents the rule source).
490
491    Returns
492    -------
493    dict
494        Each criterion's inputs and pass flag, and the overall ``pass``.
495
496    Examples
497    --------
498    >>> r = sanity_criteria([0.2, 0.25], 0.18, [0.05, 0.25], {})
499    >>> r['C1']['pass'], r['C2']['pass'], r['C3']['pass'], r['pass']
500    (True, True, True, True)
501    >>> sanity_criteria([0.5], 0.18, [0.05, 0.25], {})['C3']['pass']
502    False
503    """
504    late = [float(v) for v in late_predictions]
505    low, high = min(training_targets), max(training_targets)
506    width = high - low
507    bounds = (low - 0.5 * width, high + 0.5 * width)
508    c1 = dict(
509        mean_late_prediction=float(np.mean(late)),
510        prediction_at_100k=float(prediction_at_100k),
511        **{'pass': bool(np.mean(late) >= prediction_at_100k)},
512    )
513    c2 = dict(
514        bounds=[-0.05, 1.0],
515        min=min(late),
516        max=max(late),
517        **{'pass': all(-0.05 <= v <= 1.0 for v in late)},
518    )
519    c3 = dict(
520        bounds=list(bounds),
521        training_target_range=[low, high],
522        min=min(late),
523        max=max(late),
524        **{'pass': all(bounds[0] <= v <= bounds[1] for v in late)},
525    )
526    return dict(C1=c1, C2=c2, C3=c3, **{'pass': c1['pass'] and c2['pass'] and c3['pass']})

251def build_checkpoints(summaries, protocol):
252    """Aggregate eligible recordings into checkpoints with both input sets.
253
254    Parameters
255    ----------
256    summaries : list of dict
257        Output of :func:`extract_specimen` for one specimen.
258    protocol : dict
259        Output of :func:`load_protocol`.
260
261    Returns
262    -------
263    checkpoints : list of dict
264        Every checkpoint with at least one eligible recording, in cycle order,
265        with ``relative_rms``/``relative_ptp`` (pair 1 -> 7) and
266        ``mp_relative_rms``/``mp_relative_ptp`` (median of 36 per-pair ratios).
267        ``published_index`` is ``None`` when unlabelled or labels were not read.
268    pair_ratios : list of dict
269        Per-checkpoint, per-pair ratios behind the multi-path features.
270    reference : dict
271        Reference cycle, files and denominators.
272
273    Raises
274    ------
275    ValueError
276        If no recording is eligible, labels conflict within a checkpoint, or a
277        reference denominator is not positive.
278    """
279    eligible = [s for s in summaries if s['eligible']]
280    require(bool(eligible), 'No eligible recording')
281    specimen = eligible[0]['specimen']
282    groups = defaultdict(list)
283    for s in eligible:
284        groups[s['cycles']].append(s)
285    grid = pairs(protocol)
286    base = (
287        protocol['eligibility']['baseline_path']['actuator'],
288        protocol['eligibility']['baseline_path']['sensor'],
289    )
290    medians = {}
291    for cycle, group in groups.items():
292        medians[cycle] = {
293            p: (
294                float(np.median([s['rms'][p] for s in group])),
295                float(np.median([s['ptp'][p] for s in group])),
296            )
297            for p in grid
298        }
299    ref_cycle = min(groups)  # earliest eligible checkpoint; labels never consulted
300    ref = medians[ref_cycle]
301    require(all(v > 0 for p in grid for v in ref[p]), 'Invalid reference denominator')
302    reference = dict(
303        specimen=specimen,
304        cycles=int(ref_cycle),
305        filenames='; '.join(s['filename'] for s in groups[ref_cycle]),
306        rms_denominator=ref[base][0],
307        ptp_denominator=ref[base][1],
308    )
309    checkpoints, pair_ratios = [], []
310    for cycle in sorted(groups):
311        group = groups[cycle]
312        labels = {s['published_index'] for s in group if s['published_index'] is not None}
313        require(len(labels) <= 1, f'Conflicting published labels: {specimen}, {cycle}')
314        rms_ratio = {p: medians[cycle][p][0] / ref[p][0] for p in grid}
315        ptp_ratio = {p: medians[cycle][p][1] / ref[p][1] for p in grid}
316        checkpoints.append(
317            dict(
318                specimen=specimen,
319                cycles=int(cycle),
320                published_index=labels.pop() if labels else None,
321                eligible_repeats=len(group),
322                filenames='; '.join(s['filename'] for s in group),
323                load_min_kips=min(s['load_kips'] for s in group),
324                load_max_kips=max(s['load_kips'] for s in group),
325                is_reference=cycle == ref_cycle,
326                reference_cycles=int(ref_cycle),
327                mean_centred_rms=medians[cycle][base][0],
328                peak_to_peak=medians[cycle][base][1],
329                relative_rms=medians[cycle][base][0] / ref[base][0],
330                relative_ptp=medians[cycle][base][1] / ref[base][1],
331                mp_relative_rms=float(np.median(list(rms_ratio.values()))),
332                mp_relative_ptp=float(np.median(list(ptp_ratio.values()))),
333                log10_cycles=float(np.log10(cycle)),
334            )
335        )
336        for p in grid:
337            pair_ratios.append(
338                dict(
339                    specimen=specimen,
340                    cycles=int(cycle),
341                    actuator=int(p[0]),
342                    sensor=int(p[1]),
343                    rms_ratio=rms_ratio[p],
344                    ptp_ratio=ptp_ratio[p],
345                )
346            )
347    return checkpoints, pair_ratios, reference

480
481    Parameters
482    ----------
483    late_predictions : sequence of float
484        Frozen-model predictions at the unlabelled late L1S11 checkpoints.

526    )
527    return dict(C1=c1, C2=c2, C3=c3, **{'pass': c1['pass'] and c2['pass'] and c3['pass']})
