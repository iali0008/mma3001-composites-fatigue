exploratory_regression

Rebuild the two-specimen experiment from the external NASA dataset. Dependencies: numpy, scipy, matplotlib. No tuning, imputation, or source writes. Reference measurements are chronological anchors, not undamaged baselines.

The reference checkpoint is chosen label-blind: it is the earliest checkpoint with an acquisition-eligible recording, whether or not that checkpoint carries a published label. Fitting and scoring use only labelled, non-reference checkpoints. Specimen constants come from config/experiment_v1.json.

Example

python scripts/exploratory_regression.py --data-root "data/2.+Composites/2. Composites"

 51def is_jump(specimen, cycle, experiment):
 52    """Return whether a checkpoint is the retained published-label jump.
 53
 54    Parameters
 55    ----------
 56    specimen : str
 57        Specimen name.
 58    cycle : float or None
 59        Cycle count.
 60    experiment : dict
 61        Configuration from :func:`common.load_experiment_v1`.
 62
 63    Returns
 64    -------
 65    bool
 66        ``True`` for the configured jump checkpoint (L1S12 at 30,000 cycles).
 67    """
 68    jump = experiment['integrity']['retained_jump']
 69    return specimen == jump['specimen'] and cycle == jump['cycles']

 72def assess_recording(coupon, specimen, filename, experiment):
 73    """Screen one PZT recording and compute its receiver features.
 74
 75    Parameters
 76    ----------
 77    coupon : dict
 78        The ``coupon`` structure of one MAT file (``simplify_cells=True``).
 79    specimen : str
 80        Specimen name, a key of ``experiment['specimens']``.
 81    filename : str
 82        MAT file name, recorded for provenance.
 83    experiment : dict
 84        Configuration from :func:`common.load_experiment_v1`.
 85
 86    Returns
 87    -------
 88    dict
 89        Exported fields (historical column order) plus private fields starting
 90        with ``_``: ``_acquisition_eligible`` (all rules except the label rule)
 91        and ``_rms``/``_ptp`` (features of any acquisition-eligible recording).
 92        ``eligible`` keeps its historical meaning: acquisition-eligible *and*
 93        labelled. Public ``peak_to_peak``/``mean_centred_rms`` are filled only for
 94        ``eligible`` rows, as in the recorded tables.
 95
 96    Notes
 97    -----
 98    The acquisition decision never depends on the label; the label is read only
 99    to report it and to add the label reason to ``exclusion_reasons``.
100    """
101    settings = experiment['specimens'][specimen]
102    path_rule = experiment['path']
103    cycle, load = scalar(coupon.get('cycles', [])), scalar(coupon.get('load', []))
104    label = scalar(coupon.get('straingage_data', {}).get('stiffness_degradation', []))
105    r = dict(
106        specimen=specimen,
107        filename=filename,
108        cycles=cycle,
109        load_kips=load,
110        published_index=label,
111        condition=coupon.get('condition', ''),
112        published_jump_flag=is_jump(specimen, cycle, experiment),
113        peak_to_peak=None,
114        mean_centred_rms=None,
115    )
116    reasons = []
117    if cycle is None or cycle <= 0:
118        reasons.append('not a positive-cycle checkpoint')
119    if label is None:
120        reasons.append(LABEL_REASON)
121    if str(r['condition']).strip().casefold() != experiment['condition']:
122        reasons.append('condition is not Loaded')
123    low, high = experiment['load_window_kips']
124    if load is None or not low <= load <= high:
125        reasons.append(f'load outside {low}-{high} kips')
126    paths = coupon.get('path_data', [])
127    if isinstance(paths, dict):
128        paths = [paths]
129    wanted = (path_rule['actuator'], path_rule['sensor'], path_rule['frequency'])
130    matches = [
131        p for p in paths if (p.get('actuator'), p.get('sensor'), p.get('frequency')) == wanted
132    ]
133    sensor = None
134    if len(matches) != 1:
135        reasons.append('requested path missing or ambiguous')
136    else:
137        p = matches[0]
138        for field, required in [
139            ('gain', settings['gain']),
140            ('amplitude', settings['amplitude']),
141            ('sampling_rate', experiment['sampling_rate']),
142        ]:
143            r[field] = scalar(p.get(field, []))
144            if r[field] != required:
145                reasons.append(f'{field} != {required}')
146        for field in ('signal_actuator', 'signal_sensor'):
147            x = np.asarray(p.get(field, []), dtype=float).ravel()
148            r[field + '_length'] = len(x)
149            r[field + '_nonfinite'] = bool(not np.isfinite(x).all())
150            r[field + '_constant'] = bool(len(x) and np.isfinite(x).all() and np.all(x == x[0]))
151            if len(x) != experiment['signal_length']:
152                reasons.append(f"{field} length != {experiment['signal_length']}")
153        sensor = np.asarray(p.get('signal_sensor', []), dtype=float).ravel()
154    acquisition_ok = not [s for s in reasons if s != LABEL_REASON]
155    r['_acquisition_eligible'] = acquisition_ok
156    r['_rms'] = r['_ptp'] = None
157    if acquisition_ok and np.isfinite(sensor).all():
158        r['_rms'], r['_ptp'] = waveform_features(sensor)
159        if not reasons:
160            r['mean_centred_rms'], r['peak_to_peak'] = r['_rms'], r['_ptp']
161    r['eligible'] = not reasons
162    r['exclusion_reasons'] = '; '.join(reasons)
163    return r

166def exclusion_counts(specimen, rows):
167    """Count exclusion reasons for one specimen, in order of first appearance.
168
169    Parameters
170    ----------
171    specimen : str
172        Specimen name.
173    rows : list of dict
174        Outputs of :func:`assess_recording` for that specimen.
175
176    Returns
177    -------
178    list of dict

180    """

182    return [dict(specimen=specimen, reason=k, count=v) for k, v in counts.items()]

185def select_reference(rows):
186    """Choose the reference checkpoint without reading labels.
187
188    Parameters
189    ----------
190    rows : list of dict
191        Outputs of :func:`assess_recording` for one specimen, in file order.
192
193    Returns
194    -------
195    dict
196        ``cycles`` (int), ``filenames`` (``'; '``-joined, file order),
197        ``rms_denominator`` and ``ptp_denominator`` (medians across the
198        reference checkpoint's acquisition-eligible recordings).
199
200    Raises
201    ------
202    ValueError
203        If no recording is acquisition-eligible or a denominator is not positive.
204
205    Notes
206    -----
207    Only ``_acquisition_eligible``, ``cycles`` and the private features are read,
208    so an earlier unlabelled checkpoint would be chosen over a labelled one.
209    """
210    candidates = [r for r in rows if r['_acquisition_eligible'] and r['_rms'] is not None]
211    require(bool(candidates), 'No acquisition-eligible reference recording')
212    first = min(r['cycles'] for r in candidates)
213    group = [r for r in candidates if r['cycles'] == first]
214    reference = dict(
215        cycles=int(first),
216        filenames='; '.join(r['filename'] for r in group),
217        rms_denominator=float(np.median([r['_rms'] for r in group])),
218        ptp_denominator=float(np.median([r['_ptp'] for r in group])),
219    )
220    require(
221        reference['rms_denominator'] > 0 and reference['ptp_denominator'] > 0,
222        'Invalid reference denominator',
223    )
224    return reference

227def summarise_checkpoint(specimen, cycle, repeats, experiment):
228    """Combine the eligible repeats of one labelled checkpoint.
229
230    Parameters
231    ----------
232    specimen : str
233        Specimen name.
234    cycle : float
235        Cycle count shared by ``repeats``.
236    repeats : list of dict
237        Eligible (labelled) recordings at that checkpoint.
238    experiment : dict
239        Configuration from :func:`common.load_experiment_v1`.
240
241    Returns
242    -------
243    dict
244        Checkpoint row with the feature medians across repeats.
245
246    Raises
247    ------
248    ValueError
249        If the repeats carry conflicting labels; labels are never averaged.
250    """
251    settings = experiment['specimens'][specimen]
252    labels = set(r['published_index'] for r in repeats)
253    require(len(labels) == 1, f'Conflicting published labels: {specimen}, {cycle}')
254    cp = dict(
255        specimen=specimen,
256        cycles=int(cycle),
257        published_index=labels.pop(),
258        eligible_repeats=len(repeats),
259        filenames='; '.join(r['filename'] for r in repeats),
260        load_min_kips=min(r['load_kips'] for r in repeats),
261        load_max_kips=max(r['load_kips'] for r in repeats),
262        gain=settings['gain'],
263        excitation_setting=settings['amplitude'],
264        sampling_rate_Hz=experiment['sampling_rate'],
265        sample_start=0,
266        sample_end_inclusive=experiment['signal_length'] - 1,
267        published_jump_flag=any(r['published_jump_flag'] for r in repeats),
268    )
269    for feature in FEATURES:
270        values = [r[feature] for r in repeats if r[feature] is not None]
271        cp[feature] = float(np.median(values)) if values else None
272        cp[feature + '_valid_repeats'] = len(values)
273    return cp

276def build_checkpoints(specimen, rows, experiment):
277    """Build labelled checkpoint rows and their label-blind reference.
278
279    Parameters
280    ----------
281    specimen : str
282        Specimen name.
283    rows : list of dict
284        Outputs of :func:`assess_recording` for that specimen.
285    experiment : dict
286        Configuration from :func:`common.load_experiment_v1`.
287
288    Returns
289    -------
290    checkpoints : list of dict
291        Labelled checkpoints in cycle order with relative features. A reference
292        checkpoint without a label is included with ``published_index=None``.
293    reference : dict
294        Output of :func:`select_reference` plus ``specimen``.
295
296    Raises
297    ------
298    ValueError
299        If there are no labelled checkpoints or a checkpoint feature is missing.
300    """
301    groups = defaultdict(list)
302    for r in rows:
303        if r['eligible']:
304            groups[r['cycles']].append(r)
305    cps = [
306        summarise_checkpoint(specimen, c, reps, experiment) for c, reps in sorted(groups.items())
307    ]
308    require(bool(cps), f'No eligible checkpoints: {specimen}')
309    ref = select_reference(rows)
310    reference = dict(specimen=specimen, **ref)
311    if not any(cp['cycles'] == ref['cycles'] for cp in cps):
312        # Unlabelled reference: keep its features for normalisation, never for scoring.
313        group = [r for r in rows if r['_acquisition_eligible'] and r['cycles'] == ref['cycles']]
314        cp = summarise_checkpoint(
315            specimen,
316            ref['cycles'],
317            [
318                dict(r, published_index=None, mean_centred_rms=r['_rms'], peak_to_peak=r['_ptp'])
319                for r in group
320            ],
321            experiment,
322        )
323        cps = sorted(cps + [cp], key=lambda c: c['cycles'])
324    for cp in cps:
325        require(
326            cp['mean_centred_rms'] is not None and cp['peak_to_peak'] is not None,
327            'Nonfinite checkpoint features',
328        )
329        cp['is_reference'] = cp['cycles'] == ref['cycles']
330        cp['reference_cycles'] = ref['cycles']
331        cp['relative_rms'] = cp['mean_centred_rms'] / ref['rms_denominator']
332        cp['relative_ptp'] = cp['peak_to_peak'] / ref['ptp_denominator']
333        cp['log10_cycles'] = float(np.log10(cp['cycles']))
334    return cps, reference

337def fit_ols(train, test, fields):
338    """Fit OLS with an intercept on ``train`` and predict ``test``.
339
340    Parameters
341    ----------
342    train, test : list of dict
343        Checkpoint rows; ``published_index`` is read from ``train`` only.
344    fields : list of str
345        Predictor columns.
346
347    Returns
348    -------
349    beta : numpy.ndarray
350        Intercept followed by slopes.
351    prediction : numpy.ndarray
352        Unclipped predictions for ``test``.
353    condition_number : float
354        Ratio of the largest to smallest singular value of the design matrix.
355
356    Raises
357    ------
358    ValueError
359        If the design matrix is rank-deficient.
360    AssertionError
361        If the least-squares solution disagrees with an independent QR solve
362        (``numpy.testing`` checks are not removed by ``python -O``).
363    """
364    X = np.array([[1.0] + [r[f] for f in fields] for r in train])
365    y = np.array([r['published_index'] for r in train])
366    Xtest = np.array([[1.0] + [r[f] for f in fields] for r in test])
367    beta, _, rank, singular = np.linalg.lstsq(X, y, rcond=None)
368    require(rank == X.shape[1], 'Rank-deficient design')
369    q, upper = np.linalg.qr(X, mode='reduced')
370    np.testing.assert_allclose(beta, np.linalg.solve(upper, q.T @ y), rtol=1e-9, atol=1e-10)
371    return beta, Xtest @ beta, float(singular[0] / singular[-1])

374def evaluate_transfer(checkpoints, experiment):
375    """Fit every model on one specimen and score it on the other, both directions.
376
377    Parameters
378    ----------
379    checkpoints : list of dict
380        Output of :func:`build_checkpoints` for both specimens.
381    experiment : dict
382        Configuration from :func:`common.load_experiment_v1`.
383
384    Returns
385    -------
386    predictions, metrics, coefficients : list of dict
387        Per-checkpoint predictions, MAE/RMSE per model and direction, and fitted
388        coefficients. Reference and unlabelled checkpoints are never used.
389    """
390    predictions, metrics, coefficients = [], [], []
391    usable = [r for r in checkpoints if not r['is_reference'] and r['published_index'] is not None]
392    for train_name, test_name in experiment['directions']:
393        train = [r for r in usable if r['specimen'] == train_name]
394        test = [r for r in usable if r['specimen'] == test_name]
395        for model, fields in MODELS.items():
396            beta, pred, condition = fit_ols(train, test, fields)
397            observed = np.array([r['published_index'] for r in test])
398            residual = pred - observed
399            metrics.append(
400                dict(
401                    train_specimen=train_name,
402                    test_specimen=test_name,
403                    model=model,
404                    train_checkpoints=len(train),
405                    test_checkpoints=len(test),
406                    MAE=float(np.mean(abs(residual))),
407                    RMSE=float(np.sqrt(np.mean(residual**2))),
408                )
409            )
410            coefficients.append(
411                dict(
412                    train_specimen=train_name,
413                    test_specimen=test_name,
414                    model=model,
415                    intercept=float(beta[0]),
416                    terms='; '.join(fields),
417                    slopes='; '.join(format(b, '.17g') for b in beta[1:]),
418                    design_condition_number=condition,
419                )
420            )
421            for r, prediction in zip(test, pred):
422                predictions.append(
423                    dict(
424                        train_specimen=train_name,
425                        test_specimen=test_name,
426                        model=model,
427                        cycles=r['cycles'],
428                        observed_index=r['published_index'],
429                        predicted_index=float(prediction),
430                        residual=float(prediction - r['published_index']),
431                        published_jump_flag=r['published_jump_flag'],
432                    )
433                )
434    return predictions, metrics, coefficients

437def check_integrity(recordings, checkpoints, references, predictions, experiment):
438    """Check the configured label exclusion, retained jump and reference exclusion.
439
440    Parameters
441    ----------
442    recordings, checkpoints, references, predictions : list of dict
443        Workflow outputs.
444    experiment : dict
445        Configuration from :func:`common.load_experiment_v1`.
446
447    Raises
448    ------
449    ValueError
450        If the missing label was filled, the jump was dropped or changed, or a
451        reference checkpoint was scored.
452    """
453    missing = experiment['integrity']['missing_label']
454    jump = experiment['integrity']['retained_jump']
455    key = (missing['specimen'], missing['cycles'])
456    require(
457        not any(
458            (r['specimen'], r['cycles']) == key and r['published_index'] is not None
459            for r in checkpoints
460        ),
461        'A missing label was filled',
462    )
463    require(
464        all(
465            r['published_index'] is None for r in recordings if (r['specimen'], r['cycles']) == key
466        ),
467        'Missing-label recordings gained a label',
468    )
469    require(
470        any(
471            r['specimen'] == jump['specimen']
472            and r['cycles'] == jump['cycles']
473            and r['published_index'] == jump['published_index']
474            for r in checkpoints
475        ),
476        'Published jump not retained',
477    )
478    ref_cycles = {r['specimen']: r['cycles'] for r in references}
479    require(
480        all(p['cycles'] != ref_cycles[p['test_specimen']] for p in predictions),
481        'A reference checkpoint was scored',
482    )

485def extract(data_root, experiment, workers=1):
486    """Read, screen and summarise the PZT files of both specimens.
487
488    Parameters
489    ----------
490    data_root : pathlib.Path
491        The extracted ``2. Composites`` directory.
492    experiment : dict
493        Configuration from :func:`common.load_experiment_v1`.
494    workers : int, optional
495        MAT prefetch workers (1 = serial).
496
497    Returns
498    -------
499    recordings, checkpoints, references, exclusions : list of dict
500        Workflow tables.
501    hashes : dict
502        SHA-256 of every MAT file read, keyed relative to ``data_root``.
503    """
504    recordings, checkpoints, references, hashes, exclusions = [], [], [], {}, []
505    for specimen, settings in experiment['specimens'].items():
506        files = sorted((data_root / settings['folder'] / 'PZT-data').glob(specimen + '_*.mat'))
507        local = []
508        for path, source_digest, coupon in iter_coupons(files, workers):
509            hashes[path.relative_to(data_root).as_posix()] = source_digest
510            local.append(assess_recording(coupon, specimen, path.name, experiment))
511        recordings.extend(local)
512        exclusions.extend(exclusion_counts(specimen, local))
513        cps, reference = build_checkpoints(specimen, local, experiment)
514        references.append(reference)
515        checkpoints.extend(cps)
516    return recordings, checkpoints, references, exclusions, hashes

551def plot_predictions(predictions, metrics, experiment, path):
552    """Plot held-out predictions against observations for every model and direction.
553
554    Parameters
555    ----------
556    predictions, metrics : list of dict
557        Outputs of :func:`evaluate_transfer`.
558    experiment : dict
559        Configuration from :func:`common.load_experiment_v1`.
560    path : pathlib.Path
561        Destination PNG.
562    """
563    fig, axes = plt.subplots(2, 3, figsize=(14, 9), sharex=True, sharey=True)
564    all_values = [p[k] for p in predictions for k in ('observed_index', 'predicted_index')]
565    lo, hi = min(all_values), max(all_values)
566    pad = (hi - lo) * 0.09
567    limits = (lo - pad, hi + pad)
568    titles = ['log10(cycle count)', 'Relative RMS', 'Relative RMS + relative peak-to-peak']
569    for row, (train_name, test_name) in enumerate(experiment['directions']):
570        for col, (model, title) in enumerate(zip(MODELS, titles)):
571            ax = axes[row, col]
572            ps = [
573                p for p in predictions if p['train_specimen'] == train_name and p['model'] == model
574            ]
575            ax.plot(limits, limits, '--', color='#777777', lw=1, label='Perfect agreement')
576            ax.scatter(
577                [p['observed_index'] for p in ps],
578                [p['predicted_index'] for p in ps],
579                color='#24628a',
580                s=30,
581            )
582            for p in ps:
583                if p['published_jump_flag']:
584                    ax.scatter(
585                        p['observed_index'],
586                        p['predicted_index'],
587                        marker='D',
588                        s=65,
589                        facecolors='none',
590                        edgecolors='#b45e13',
591                        linewidths=1.5,
592                    )
593            m = next(
594                m for m in metrics if m['train_specimen'] == train_name and m['model'] == model
595            )
596            ax.set(
597                title=f'{train_name} -> {test_name}\n{title}',
598                xlim=limits,
599                ylim=limits,
600                xlabel='Observed published index',
601                ylabel='Predicted index',
602            )
603            ax.text(

606                f"MAE {m['MAE']:.4f}\nRMSE {m['RMSE']:.4f}",

608                va='top',
609                fontsize=9,
610            )
611            ax.grid(True, color='#e0e0e0', lw=0.5)
612            ax.spines[['top', 'right']].set_visible(False)
613            ax.set_aspect('equal', adjustable='box')
614    fig.suptitle('Exploratory two-specimen OLS comparison | held-out checkpoints only', fontsize=14)
615    fig.text(
616        0.07,
617        0.02,
618        'Each model includes an intercept. References are excluded from fitting and '

622        fontsize=9,

629def summary_text(recordings, checkpoints, references, metrics, experiment):
630    """Compose the plain-text run summary written to ``README.txt``.
631
632    Parameters

338def fit_ols(train, test, fields):
339    """Fit OLS with an intercept on ``train`` and predict ``test``.
340
341    Parameters
342    ----------
343    train, test : list of dict
344        Checkpoint rows; ``published_index`` is read from ``train`` only.
345    fields : list of str
346        Predictor columns.
347
348    Returns
349    -------
350    beta : numpy.ndarray
351        Intercept followed by slopes.
352    prediction : numpy.ndarray
353        Unclipped predictions for ``test``.
354    condition_number : float
355        Ratio of the largest to smallest singular value of the design matrix.
356
357    Raises
358    ------
359    ValueError
360        If the design matrix is rank-deficient.
361    AssertionError
362        If the least-squares solution disagrees with an independent QR solve
363        (``numpy.testing`` checks are not removed by ``python -O``).
364    """
365    X = np.array([[1.0] + [r[f] for f in fields] for r in train])
366    y = np.array([r['published_index'] for r in train])
367    Xtest = np.array([[1.0] + [r[f] for f in fields] for r in test])
368    beta, _, rank, singular = np.linalg.lstsq(X, y, rcond=None)
369    require(rank == X.shape[1], 'Rank-deficient design')
370    q, upper = np.linalg.qr(X, mode='reduced')
371    np.testing.assert_allclose(beta, np.linalg.solve(upper, q.T @ y), rtol=1e-9, atol=1e-10)
372    return beta, Xtest @ beta, float(singular[0] / singular[-1])
